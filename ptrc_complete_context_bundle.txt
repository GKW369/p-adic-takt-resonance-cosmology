import os

def bundle_repository():
    # Ermittle das Verzeichnis, in dem dieses Skript liegt
    current_script_dir = os.path.dirname(os.path.abspath(__file__))
    
    # DYNAMISCHE ROOT-ERKENNUNG:
    # Wenn das Skript im Unterordner "src" liegt, ist das Hauptverzeichnis eine Ebene höher.
    if os.path.basename(current_script_dir) == "src":
        root_dir = os.path.dirname(current_script_dir)
    else:
        root_dir = current_script_dir
        
    output_filename = "ptrc_complete_context_bundle.txt"
    output_file = os.path.join(root_dir, output_filename)
    
    # Erlaubte Datei-Endungen für den KI-Kontext
    valid_extensions = (".py", ".md", ".txt", ".c", ".h")
    
    # Hauptordner im Root-Verzeichnis scannen
    target_root_folders = ["src", "paper", "experimental", "embedded"]
    
    print(f"[ START ] Erzeuge das globale pTRC-Kontext-Bundle im Hauptverzeichnis: {root_dir}")
    
    # Listen, um die gefundenen Pfade für das Inhaltsverzeichnis zu sammeln
    files_to_bundle = []
    
    # Schritt 1: Vorscan für das Inhaltsverzeichnis (README direkt im Root)
    readme_path = os.path.join(root_dir, "README.md")
    if os.path.exists(readme_path):
        files_to_bundle.append("README.md")
        
    for root_folder in target_root_folders:
        root_folder_path = os.path.join(root_dir, root_folder)
        if not os.path.exists(root_folder_path):
            continue
            
        for dirpath, _, filenames in os.walk(root_folder_path):
            for filename in sorted(filenames):
                if filename.endswith(valid_extensions):
                    if filename == output_filename:
                        continue
                    file_path = os.path.join(dirpath, filename)
                    # Relativen Pfad immer basierend auf dem Hauptverzeichnis berechnen
                    relative_path = os.path.relpath(file_path, root_dir)
                    files_to_bundle.append(relative_path)

    # Schritt 2: Schreiben des Bundles (Inhaltsverzeichnis zuerst, dann die Inhalte)
    with open(output_file, "w", encoding="utf-8") as outfile:
        # Header und Inhaltsverzeichnis in die Datei schreiben
        outfile.write("======================================================================\n")
        outfile.write("pTRC REPOSITORY COMPLETE CONTEXT BUNDLE\n")
        outfile.write("======================================================================\n\n")
        outfile.write("--- TABLE OF CONTENTS ---\n")
        for idx, rel_path in enumerate(files_to_bundle, 1):
            outfile.write(f"{idx:02d}. {rel_path}\n")
        outfile.write("----------------------------------------------------------------------\n\n\n")
        
        # Jetzt die Dateien nacheinander einlesen und anhängen
        for rel_path in files_to_bundle:
            full_path = os.path.join(root_dir, rel_path)
            
            outfile.write("======================================================================\n")
            outfile.write(f"--- START OF REPOSITORY FILE: {rel_path} ---\n")
            outfile.write("======================================================================\n\n")
            
            with open(full_path, "r", encoding="utf-8") as infile:
                outfile.write(infile.read())
                
            outfile.write("\n\n")
            print(f"        Gebündelt und indexiert: {rel_path}")
                        
    print(f"\n[ OK ] Das komplette Repository wurde vereint (inkl. Index): {output_file}")

if __name__ == "__main__":
    bundle_repository()

### 1. Zero-Float Inference Transformation
The underlying cryptographic core mapped in `src/00_ptrc_crypt_demo.py` executing inside the cyclically closed primorial integer ring (\(\mathbb{Z}/210\mathbb{Z} \times \mathbb{Z}/24\mathbb{Z}\)) maps inputs losslessly without floating-point pollution:

```python
def ptrc_crypt_engine_bytes(data: bytes, mode: str = "ENCRYPT") -> bytes:
    MOD_BASE, CONTROL_TAKT, KEY_SHIFT = 210, 24, 48
    output = bytearray(len(data))
    for i, byte_val in enumerate(data):
        takt_offset = ((i * CONTROL_TAKT) + KEY_SHIFT) % MOD_BASE
        if mode == "ENCRYPT":
            output[i] = (byte_val + takt_offset) % 256
```
